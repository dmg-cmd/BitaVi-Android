package com.example.location.manager

import android.annotation.SuppressLint
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build
import android.util.Log
import com.example.location.receiver.ActivityTransitionReceiver
import com.google.android.gms.location.ActivityRecognition
import com.google.android.gms.location.ActivityTransition
import com.google.android.gms.location.ActivityTransitionRequest
import com.google.android.gms.location.DetectedActivity

class ActivityRecognitionManager(private val context: Context) {

    companion object {
        private const val REQUEST_CODE = 4004
        private const val TAG = "ACTIVITY_REC_MANAGER"
    }

    private val pendingIntent: PendingIntent by lazy {
        val intent = Intent(context, ActivityTransitionReceiver::class.java)
        val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            PendingIntent.FLAG_MUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        } else {
            PendingIntent.FLAG_UPDATE_CURRENT
        }
        PendingIntent.getBroadcast(context, REQUEST_CODE, intent, flags)
    }

    private fun buildTransitionRequest(): ActivityTransitionRequest {
        val transitions = mutableListOf<ActivityTransition>()

        val targetActivities = listOf(
            DetectedActivity.IN_VEHICLE,
            DetectedActivity.ON_BICYCLE,
            DetectedActivity.RUNNING,
            DetectedActivity.WALKING,
            DetectedActivity.STILL
        )

        for (activity in targetActivities) {
            transitions.add(ActivityTransition.Builder()
                .setActivityType(activity)
                .setActivityTransition(ActivityTransition.ACTIVITY_TRANSITION_ENTER)
                .build())
            transitions.add(ActivityTransition.Builder()
                .setActivityType(activity)
                .setActivityTransition(ActivityTransition.ACTIVITY_TRANSITION_EXIT)
                .build())
        }

        return ActivityTransitionRequest(transitions)
    }

    @SuppressLint("MissingPermission")
    fun startTrackingTransitions() {
        ActivityRecognition.getClient(context)
            .requestActivityTransitionUpdates(buildTransitionRequest(), pendingIntent)
            .addOnSuccessListener {
                Log.i(TAG, "Suscripcion exitosa al arranque automatico por transiciones.")
            }
            .addOnFailureListener { e ->
                Log.w(TAG, "Error al suscribirse al reconocimiento de actividad: ${e.message}")
            }
    }

    fun stopTrackingTransitions() {
        ActivityRecognition.getClient(context)
            .removeActivityTransitionUpdates(pendingIntent)
            .addOnSuccessListener {
                Log.i(TAG, "Remocion de actualizaciones automaticas completada en limpio.")
            }
            .addOnFailureListener { e ->
                Log.w(TAG, "Fallo al remover el monitoreo de transiciones: ${e.message}")
            }
    }
}

package com.aistudio.localizador.ui

import android.content.Context
import android.net.Uri
import android.provider.ContactsContract

class ContactUseCase(private val context: Context) {

    data class ContactDetails(
        val name: String,
        val phone: String,
        val email: String,
        val address: String
    )

    suspend fun getContactDetails(contactUri: Uri): ContactDetails {
        var name = ""
        var phone = ""
        var email = ""
        var address = ""

        val contentResolver = context.contentResolver
        contentResolver.query(contactUri, null, null, null, null)?.use { cursor ->
            if (cursor.moveToFirst()) {
                val id = cursor.getString(cursor.getColumnIndexOrThrow(android.provider.ContactsContract.Contacts._ID))
                name = cursor.getString(cursor.getColumnIndexOrThrow(android.provider.ContactsContract.Contacts.DISPLAY_NAME))

                if (cursor.getInt(cursor.getColumnIndexOrThrow(android.provider.ContactsContract.Contacts.HAS_PHONE_NUMBER)) > 0) {
                    contentResolver.query(
                        android.provider.ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
                        null,
                        "${android.provider.ContactsContract.CommonDataKinds.Phone.CONTACT_ID} = ?",
                        arrayOf(id),
                        null
                    )?.use { pCursor ->
                        if (pCursor.moveToFirst()) {
                            phone = pCursor.getString(pCursor.getColumnIndexOrThrow(android.provider.ContactsContract.CommonDataKinds.Phone.NUMBER))
                        }
                    }
                }

                contentResolver.query(
                    android.provider.ContactsContract.CommonDataKinds.Email.CONTENT_URI,
                    null,
                    "${android.provider.ContactsContract.CommonDataKinds.Email.CONTACT_ID} = ?",
                    arrayOf(id),
                    null
                )?.use { eCursor ->
                    if (eCursor.moveToFirst()) {
                        email = eCursor.getString(eCursor.getColumnIndexOrThrow(android.provider.ContactsContract.CommonDataKinds.Email.ADDRESS))
                    }
                }

                contentResolver.query(
                    android.provider.ContactsContract.CommonDataKinds.StructuredPostal.CONTENT_URI,
                    null,
                    "${android.provider.ContactsContract.CommonDataKinds.StructuredPostal.CONTACT_ID} = ?",
                    arrayOf(id),
                    null
                )?.use { aCursor ->
                    if (aCursor.moveToFirst()) {
                        address = aCursor.getString(aCursor.getColumnIndexOrThrow(android.provider.ContactsContract.CommonDataKinds.StructuredPostal.FORMATTED_ADDRESS))
                    }
                }
            }
        }
        return ContactDetails(name, phone, email, address)
    }

    fun getAllContactPhones(): List<Pair<String, String>> {
        val result = mutableListOf<Pair<String, String>>()
        val contentResolver = context.contentResolver
        val uri = ContactsContract.CommonDataKinds.Phone.CONTENT_URI
        val projection = arrayOf(
            ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME,
            ContactsContract.CommonDataKinds.Phone.NUMBER
        )
        contentResolver.query(uri, projection, null, null, null)?.use { cursor ->
            val nameIdx = cursor.getColumnIndex(ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME)
            val phoneIdx = cursor.getColumnIndex(ContactsContract.CommonDataKinds.Phone.NUMBER)
            while (cursor.moveToNext()) {
                val name = if (nameIdx >= 0) cursor.getString(nameIdx).orEmpty() else ""
                val phone = if (phoneIdx >= 0) cursor.getString(phoneIdx).orEmpty() else ""
                if (phone.isNotBlank()) result.add(name to phone)
            }
        }
        return result
    }
}

package com.aistudio.localizador.ui

import android.location.Geocoder
import java.util.Locale
import java.util.concurrent.ConcurrentHashMap

object LocationParser {

    private val geoCache = ConcurrentHashMap<String, Pair<String, String>>()

    fun inferCountryCity(
        address: String,
        lat: Double,
        lon: Double,
        geocoder: Geocoder
    ): Pair<String, String> {
        val parsed = parseFromAddressString(address)
        if (parsed.first != "Desconocido") return parsed

        val cacheKey = String.format(Locale.US, "%.3f,%.3f", lat, lon)
        geoCache[cacheKey]?.let { return it }

        return try {
            @Suppress("DEPRECATION")
            val addresses = geocoder.getFromLocation(lat, lon, 1)
            if (!addresses.isNullOrEmpty()) {
                val addr = addresses[0]
                val country = addr.countryName ?: "Desconocido"
                val city = addr.locality ?: addr.adminArea ?: "Desconocida"
                Pair(country, city).also { geoCache[cacheKey] = it }
            } else {
                Pair("Desconocido", "Desconocida")
            }
        } catch (_: Exception) {
            Pair("Desconocido", "Desconocida")
        }
    }

    fun parseFromAddressString(address: String): Pair<String, String> {
        if (address.isBlank() || address.startsWith("GPS:")) {
            return Pair("Desconocido", "Desconocida")
        }
        val parts = address.split(",").map { it.trim() }.filter { it.isNotBlank() }
        if (parts.size >= 2) {
            val last = parts.last()
            val secondLast = parts[parts.size - 2]
            if (last.any { it.isLetter() } && secondLast.any { it.isLetter() }) {
                return Pair(last, secondLast)
            }
        }
        return Pair("Desconocido", "Desconocida")
    }

    fun clearCache() {
        geoCache.clear()
    }
}

package com.aistudio.localizador.util

import android.content.Context
import android.media.ToneGenerator
import android.os.Build
import android.os.VibrationEffect
import android.os.Vibrator
import android.os.VibratorManager

class AlertNotifier(private val context: Context) {

    private var toneGenerator: ToneGenerator? = null

    fun triggerProximityAlert() {
        playTone()
        vibrate()
    }

    private fun playTone() {
        try {
            if (toneGenerator == null) {
                toneGenerator = ToneGenerator(
                    AppConstants.AudioVisual.PROXIMITY_TONE,
                    ToneGenerator.MAX_VOLUME
                )
            }
            toneGenerator?.startTone(
                AppConstants.AudioVisual.PROXIMITY_TONE,
                AppConstants.AudioVisual.TONE_DURATION_MS.toInt()
            )
        } catch (_: Exception) { }
    }

    private fun vibrate() {
        try {
            val vibrator = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                val vm = context.getSystemService(Context.VIBRATOR_MANAGER_SERVICE) as VibratorManager
                vm.defaultVibrator
            } else {
                @Suppress("DEPRECATION")
                context.getSystemService(Context.VIBRATOR_SERVICE) as Vibrator
            }
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                vibrator.vibrate(
                    VibrationEffect.createWaveform(
                        AppConstants.AudioVisual.PROXIMITY_VIBRATE_PATTERN,
                        -1
                    )
                )
            } else {
                @Suppress("DEPRECATION")
                vibrator.vibrate(AppConstants.AudioVisual.PROXIMITY_VIBRATE_PATTERN, -1)
            }
        } catch (_: Exception) { }
    }

    fun release() {
        try { toneGenerator?.release() } catch (_: Exception) { }
        toneGenerator = null
    }
}
